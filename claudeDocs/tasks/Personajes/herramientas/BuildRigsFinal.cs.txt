    public sealed class Tabla { public Personaje[] personajes; }

    public sealed class Personaje
    {
        public string id; public string prefab; public string carpeta; public string prefijo; public bool guia;
        public Nodo[] nodos; public Parte[] partes;
    }

    /// <summary>
    /// Un nodo que se AÑADE. «tipo»: articulacion = pivote de tamaño 0 en «punto» con un segmento
    /// «imagen» colgado de él (con «rect»); imagen = el propio nodo lleva la Image; grupo = nodo
    /// estirado sin Image, con el pivote en «punto». «padre» es la ruta desde la raíz del prefab.
    /// Según el tipo son obligatorios «punto» (todos), «imagen» (articulacion) y «rect» (grupo); lo que
    /// el JSON omita o ponga en null queda en null, que es lo que el resto del script ya espera.
    /// </summary>
    public sealed class Nodo
    {
        public string nombre; public string tipo; public string padre; public string imagen; public string sprite;
        public float[] punto; public float[] rect;
    }

    /// <summary>Una parte que YA existe: su ruta, su sprite y el rect y pivote con que se reajusta.</summary>
    public sealed class Parte
    {
        public string nombre; public string ruta; public string sprite;
        public float[] rect; public float[] pivote;
    }

    /// <summary>
    /// Lee el texto de la tabla. Obligatorios: id, prefab, carpeta, prefijo y nodos por personaje; nombre,
    /// tipo y padre por nodo; nombre, ruta, rect y pivote por parte. «partes» ausente = lista vacía.
    /// Lanza InvalidOperationException con el motivo y el lugar exactos.
    /// </summary>
    public static Tabla ParseTable(string json)
    {
        var root = MiniJson.Parse(json) as Dictionary<string, object>;
        if (root == null)
        {
            throw new InvalidOperationException("la raíz de la tabla debe ser un objeto JSON");
        }

        var items = Items(root, "personajes", "tabla", true);
        if (items.Count == 0)
        {
            throw new InvalidOperationException("tabla: «personajes» está vacío");
        }

        var tabla = new Tabla { personajes = new Personaje[items.Count] };
        for (var i = 0; i < items.Count; i++)
        {
            tabla.personajes[i] = ToPersonaje(items[i], i);
        }

        return tabla;
    }

    private static Personaje ToPersonaje(object raw, int index)
    {
        var ctx = "personajes[" + index + "]";
        var d = AsObject(raw, ctx);
        var id = Str(d, "id", ctx, true);
        ctx = "personaje «" + id + "»";

        var p = new Personaje
        {
            id = id,
            prefab = Str(d, "prefab", ctx, true),
            carpeta = Str(d, "carpeta", ctx, true),
            prefijo = Str(d, "prefijo", ctx, true),
            guia = Flag(d, "guia", ctx),
        };

        var nodos = Items(d, "nodos", ctx, true);
        p.nodos = new Nodo[nodos.Count];
        for (var k = 0; k < nodos.Count; k++)
        {
            p.nodos[k] = ToNodo(nodos[k], ctx, k);
        }

        var partes = Items(d, "partes", ctx, false);
        p.partes = new Parte[partes.Count];
        for (var k = 0; k < partes.Count; k++)
        {
            p.partes[k] = ToParte(partes[k], ctx, k);
        }

        return p;
    }

    private static Nodo ToNodo(object raw, string owner, int index)
    {
        var ctx = owner + ", nodos[" + index + "]";
        var d = AsObject(raw, ctx);
        var nombre = Str(d, "nombre", ctx, true);
        ctx = owner + ", nodo «" + nombre + "»";

        var tipo = Str(d, "tipo", ctx, true);
        if (tipo != "articulacion" && tipo != "imagen" && tipo != "grupo")
        {
            throw new InvalidOperationException(ctx + ": tipo «" + tipo + "» desconocido (articulacion | imagen | grupo)");
        }

        return new Nodo
        {
            nombre = nombre,
            tipo = tipo,
            padre = Str(d, "padre", ctx, true),
            imagen = Str(d, "imagen", ctx, tipo == "articulacion"),
            sprite = Str(d, "sprite", ctx, false),
            punto = Numbers(d, "punto", ctx, 2, true),
            rect = Numbers(d, "rect", ctx, 4, tipo == "grupo"),
        };
    }

    private static Parte ToParte(object raw, string owner, int index)
    {
        var ctx = owner + ", partes[" + index + "]";
        var d = AsObject(raw, ctx);
        var nombre = Str(d, "nombre", ctx, true);
        ctx = owner + ", parte «" + nombre + "»";

        return new Parte
        {
            nombre = nombre,
            ruta = Str(d, "ruta", ctx, true),
            sprite = Str(d, "sprite", ctx, false),
            rect = Numbers(d, "rect", ctx, 4, true),
            pivote = Numbers(d, "pivote", ctx, 2, true),
        };
    }

    private static Dictionary<string, object> AsObject(object raw, string ctx)
    {
        var d = raw as Dictionary<string, object>;
        if (d == null)
        {
            throw new InvalidOperationException(ctx + ": debe ser un objeto JSON");
        }

        return d;
    }

    private static InvalidOperationException Missing(string ctx, string key)
    {
        return new InvalidOperationException(ctx + ": falta el campo obligatorio «" + key + "»");
    }

    /// <summary>Un texto. Ausente o null: error si es obligatorio y null si no. Obligatorio y vacío también es error.</summary>
    private static string Str(Dictionary<string, object> d, string key, string ctx, bool required)
    {
        object v;
        if (!d.TryGetValue(key, out v) || v == null)
        {
            if (required) throw Missing(ctx, key);
            return null;
        }

        var s = v as string;
        if (s == null)
        {
            throw new InvalidOperationException(ctx + ": «" + key + "» debe ser texto");
        }

        if (required && s.Length == 0)
        {
            throw new InvalidOperationException(ctx + ": el campo obligatorio «" + key + "» está vacío");
        }

        return s;
    }

    private static bool Flag(Dictionary<string, object> d, string key, string ctx)
    {
        object v;
        if (!d.TryGetValue(key, out v) || v == null)
        {
            return false;
        }

        if (!(v is bool))
        {
            throw new InvalidOperationException(ctx + ": «" + key + "» debe ser true o false");
        }

        return (bool)v;
    }

    /// <summary>Una lista de objetos. Ausente o null: error si es obligatoria y lista vacía si no.</summary>
    private static List<object> Items(Dictionary<string, object> d, string key, string ctx, bool required)
    {
        object v;
        if (!d.TryGetValue(key, out v) || v == null)
        {
            if (required) throw Missing(ctx, key);
            return new List<object>();
        }

        var list = v as List<object>;
        if (list == null)
        {
            throw new InvalidOperationException(ctx + ": «" + key + "» debe ser una lista");
        }

        return list;
    }

    /// <summary>Una lista de exactamente «length» números. Ausente o null: error si es obligatoria y null si no.</summary>
    private static float[] Numbers(Dictionary<string, object> d, string key, string ctx, int length, bool required)
    {
        object v;
        if (!d.TryGetValue(key, out v) || v == null)
        {
            if (required) throw Missing(ctx, key);
            return null;
        }

        var list = v as List<object>;
        if (list == null || list.Count != length)
        {
            throw new InvalidOperationException(ctx + ": «" + key + "» debe ser una lista de " + length + " números");
        }

        var result = new float[length];
        for (var i = 0; i < length; i++)
        {
            if (!(list[i] is double))
            {
                throw new InvalidOperationException(ctx + ": «" + key + "»[" + i + "] debe ser un número");
            }

            result[i] = (float)(double)list[i];
        }

        return result;
    }

    /// <summary>
    /// Descenso recursivo sobre el texto. Objetos → Dictionary&lt;string, object&gt;, listas →
    /// List&lt;object&gt;, números → double (cultura invariante), más string, bool y null. Estricto: sin
    /// comas finales, sin comentarios, sin caracteres de control sueltos en las cadenas. Un campo repetido
    /// en un objeto es un error (casi siempre es una errata). El BOM inicial se tolera.
    /// </summary>
    private sealed class MiniJson
    {
        private const int MaxDepth = 64;
        private readonly string text;
        private int pos;

        private MiniJson(string text) { this.text = text; }

        public static object Parse(string text)
        {
            if (text == null)
            {
                throw new InvalidOperationException("JSON ilegible: texto nulo");
            }

            var reader = new MiniJson(text);
            reader.SkipWhite();
            var value = reader.ReadValue(0);
            reader.SkipWhite();
            if (reader.pos < text.Length)
            {
                throw reader.Error("hay texto después del valor raíz");
            }

            return value;
        }

        /// <summary>Error con la línea y la columna (desde 1) de la posición actual.</summary>
        private InvalidOperationException Error(string message)
        {
            var line = 1;
            var lineStart = 0;
            for (var k = 0; k < pos && k < text.Length; k++)
            {
                if (text[k] == '\n')
                {
                    line++;
                    lineStart = k + 1;
                }
            }

            return new InvalidOperationException(
                "JSON ilegible en la línea " + line + ", columna " + (pos - lineStart + 1) + ": " + message);
        }

        private string Describe()
        {
            if (pos >= text.Length) return "el fin del texto";
            var c = text[pos];
            return c < ' ' ? "un carácter de control (U+" + ((int)c).ToString("X4") + ")" : "«" + c + "»";
        }

        private void SkipWhite()
        {
            while (pos < text.Length)
            {
                var c = text[pos];
                if (c != ' ' && c != '\t' && c != '\n' && c != '\r' && c != '﻿') break;
                pos++;
            }
        }

        private bool At(char c) { return pos < text.Length && text[pos] == c; }

        private void Expect(char c)
        {
            if (!At(c))
            {
                throw Error("se esperaba «" + c + "», no " + Describe());
            }

            pos++;
        }

        private object ReadValue(int depth)
        {
            if (depth > MaxDepth) throw Error("anidamiento excesivo");
            if (pos >= text.Length) throw Error("el texto termina donde se esperaba un valor");

            var c = text[pos];
            switch (c)
            {
                case '{': return ReadObject(depth);
                case '[': return ReadArray(depth);
                case '"': return ReadString();
                case 't': ReadLiteral("true"); return true;
                case 'f': ReadLiteral("false"); return false;
                case 'n': ReadLiteral("null"); return null;
                default:
                    if (c == '-' || (c >= '0' && c <= '9')) return ReadNumber();
                    throw Error("carácter inesperado " + Describe());
            }
        }

        private void ReadLiteral(string word)
        {
            if (string.CompareOrdinal(text, pos, word, 0, word.Length) != 0)
            {
                throw Error("se esperaba «" + word + "»");
            }

            pos += word.Length;
        }

        private Dictionary<string, object> ReadObject(int depth)
        {
            pos++; // «{»
            var result = new Dictionary<string, object>(StringComparer.Ordinal);
            SkipWhite();
            if (At('}'))
            {
                pos++;
                return result;
            }

            while (true)
            {
                SkipWhite();
                if (!At('"')) throw Error("se esperaba el nombre de un campo entre comillas, no " + Describe());
                var keyPos = pos;
                var key = ReadString();
                if (result.ContainsKey(key))
                {
                    pos = keyPos;
                    throw Error("el campo «" + key + "» está repetido");
                }

                SkipWhite();
                Expect(':');
                SkipWhite();
                result[key] = ReadValue(depth + 1);
                SkipWhite();
                if (At(',')) { pos++; continue; }
                if (At('}')) { pos++; return result; }
                throw Error("se esperaba «,» o «}», no " + Describe());
            }
        }

        private List<object> ReadArray(int depth)
        {
            pos++; // «[»
            var result = new List<object>();
            SkipWhite();
            if (At(']'))
            {
                pos++;
                return result;
            }

            while (true)
            {
                SkipWhite();
                result.Add(ReadValue(depth + 1));
                SkipWhite();
                if (At(',')) { pos++; continue; }
                if (At(']')) { pos++; return result; }
                throw Error("se esperaba «,» o «]», no " + Describe());
            }
        }

        private string ReadString()
        {
            pos++; // comilla de apertura
            var sb = new System.Text.StringBuilder();
            while (true)
            {
                if (pos >= text.Length) throw Error("cadena sin cerrar");
                var c = text[pos];
                if (c == '"')
                {
                    pos++;
                    return sb.ToString();
                }

                if (c < ' ') throw Error("carácter de control sin escapar dentro de una cadena");
                if (c != '\\')
                {
                    sb.Append(c);
                    pos++;
                    continue;
                }

                pos++; // «\»
                if (pos >= text.Length) throw Error("cadena sin cerrar");
                switch (text[pos])
                {
                    case '"': sb.Append('"'); break;
                    case '\\': sb.Append('\\'); break;
                    case '/': sb.Append('/'); break;
                    case 'b': sb.Append('\b'); break;
                    case 'f': sb.Append('\f'); break;
                    case 'n': sb.Append('\n'); break;
                    case 'r': sb.Append('\r'); break;
                    case 't': sb.Append('\t'); break;
                    case 'u':
                        int code;
                        if (pos + 4 >= text.Length
                            || !int.TryParse(text.Substring(pos + 1, 4), System.Globalization.NumberStyles.AllowHexSpecifier,
                                System.Globalization.CultureInfo.InvariantCulture, out code))
                        {
                            throw Error("«\\u» debe llevar cuatro dígitos hexadecimales");
                        }

                        sb.Append((char)code); // un par sustituto llega como dos «\uXXXX» seguidos y se recompone solo
                        pos += 4;
                        break;
                    default:
                        throw Error("secuencia de escape inválida «\\" + text[pos] + "»");
                }

                pos++;
            }
        }

        private bool AtDigit() { return pos < text.Length && text[pos] >= '0' && text[pos] <= '9'; }

        private object ReadNumber()
        {
            var start = pos;
            if (At('-')) pos++;
            if (!AtDigit()) throw Error("número mal formado: falta un dígito");
            if (At('0')) pos++;
            else while (AtDigit()) pos++;

            if (At('.'))
            {
                pos++;
                if (!AtDigit()) throw Error("número mal formado: falta un dígito tras el punto decimal");
                while (AtDigit()) pos++;
            }

            if (At('e') || At('E'))
            {
                pos++;
                if (At('+') || At('-')) pos++;
                if (!AtDigit()) throw Error("número mal formado: falta un dígito en el exponente");
                while (AtDigit()) pos++;
            }

            double value;
            if (!double.TryParse(text.Substring(start, pos - start), System.Globalization.NumberStyles.Float,
                    System.Globalization.CultureInfo.InvariantCulture, out value))
            {
                pos = start;
                throw Error("número fuera de rango");
            }

            return value;
        }
    }

    public static string Execute(string mode)
    {
        var log = new List<string>();
        try
        {
            AssetDatabase.Refresh();
            var tabla = LoadTable(log);
            switch (mode)
            {
                case "nodos":
                    foreach (var p in tabla.personajes) AddNodes(p, log);
                    break;
                case "sprites":
                    foreach (var p in tabla.personajes) ApplySprites(p, log);
                    break;
                case "clips":
                    WriteClips(tabla, log);
                    break;
                case "todo":
                    foreach (var p in tabla.personajes) AddNodes(p, log);
                    WriteClips(tabla, log);
                    break;
                case "estado":
                    foreach (var p in tabla.personajes) Report(p, log);
                    break;
                default:
                    log.Add("modo desconocido «" + mode + "»: nodos | sprites | clips | todo | estado");
                    break;
            }

            AssetDatabase.SaveAssets();
        }
        catch (Exception e)
        {
            log.Add("ERROR: " + e);
        }

        return string.Join("\n", log);
    }

    private static Tabla LoadTable(List<string> log)
    {
        var path = Path.GetFullPath(Path.Combine(Application.dataPath, "..", Json));
        Tabla tabla;
        try
        {
            tabla = ParseTable(File.ReadAllText(path, System.Text.Encoding.UTF8));
        }
        catch (InvalidOperationException e)
        {
            throw new InvalidOperationException("tabla ilegible (" + path + "): " + e.Message, e);
        }

        log.Add("tabla: " + tabla.personajes.Length + " personajes (" + path + ")");
        return tabla;
    }

    /// <summary>Rect de un nodo en el lienzo, con las coordenadas de Unity (origen abajo a la izquierda).</summary>
    private struct Box
    {
        public Vector2 Min;
        public Vector2 Size;
    }

    /// <summary>
    /// El rect de un RectTransform dentro de Lienzo, calculado solo con lo serializado. La fórmula es
    /// la de Unity: el punto de referencia de las anclas es anchorMin + pivot·(anchorMax − anchorMin),
    /// el pivote cae en referencia + anchoredPosition y el tamaño es (anchorMax − anchorMin)·padre + sizeDelta.
    /// </summary>
    private static Box BoxOf(RectTransform rt)
    {
        if (rt.name == "Lienzo")
        {
            return new Box { Min = Vector2.zero, Size = new Vector2(U, U) };
        }

        var parent = rt.parent as RectTransform;
        if (parent == null)
        {
            throw new InvalidOperationException(rt.name + " no cuelga de Lienzo");
        }

        var pb = BoxOf(parent);
        var span = rt.anchorMax - rt.anchorMin;
        var size = Vector2.Scale(span, pb.Size) + rt.sizeDelta;
        var reference = rt.anchorMin + Vector2.Scale(rt.pivot, span);
        var pivot = pb.Min + Vector2.Scale(reference, pb.Size) + rt.anchoredPosition;
        return new Box { Min = pivot - Vector2.Scale(rt.pivot, size), Size = size };
    }

    private static Vector2 PivotUp(RectTransform rt)
    {
        var b = BoxOf(rt);
        return b.Min + Vector2.Scale(rt.pivot, b.Size);
    }

    /// <summary>El pivote en coordenadas de la tabla (y hacia abajo).</summary>
    private static Vector2 PivotDown(RectTransform rt)
    {
        var p = PivotUp(rt);
        return new Vector2(p.x, U - p.y);
    }

    private static Vector2 DownToUp(Vector2 p) { return new Vector2(p.x, U - p.y); }

    private static Vector2 Pt(float[] p) { return new Vector2(p[0], p[1]); }

    private static bool Near(Vector2 a, Vector2 b) { return (a - b).sqrMagnitude < 0.0025f; }

    private static float Snap(float v) { return Mathf.Abs(v) < 1e-3f ? 0f : v; }

    /// <summary>
    /// Coloca un nodo con anclas puntuales: su pivote cae en «punto» (tabla, y hacia abajo) y, si hay
    /// «rect», el rect cuelga de ese pivote. Con un padre de tamaño 0 (un pivote de articulación) las
    /// anclas quedan en el centro y el desplazamiento absorbe la diferencia. Se asigna anchoredPosition
    /// al final porque los setters de pivot y sizeDelta la reajustan.
    /// </summary>
    private static void PlacePoint(RectTransform rt, Vector2 pointDown, float[] rect)
    {
        var pb = BoxOf((RectTransform)rt.parent);
        var p = DownToUp(pointDown);
        var anchor = new Vector2(
            pb.Size.x > 0.01f ? (p.x - pb.Min.x) / pb.Size.x : 0.5f,
            pb.Size.y > 0.01f ? (p.y - pb.Min.y) / pb.Size.y : 0.5f);
        rt.anchorMin = anchor;
        rt.anchorMax = anchor;
        if (rect == null || rect.Length < 4)
        {
            rt.pivot = new Vector2(0.5f, 0.5f);
            rt.sizeDelta = Vector2.zero;
        }
        else
        {
            var w = rect[2] - rect[0];
            var h = rect[3] - rect[1];
            rt.pivot = new Vector2((pointDown.x - rect[0]) / w, (rect[3] - pointDown.y) / h);
            rt.sizeDelta = new Vector2(w, h);
        }

        var reference = pb.Min + Vector2.Scale(anchor, pb.Size);
        var offset = p - reference;
        rt.anchoredPosition = new Vector2(Snap(offset.x), Snap(offset.y));
    }

    /// <summary>
    /// Como Part() de BuildRigs: anclas = el rect en fracciones del padre, sizeDelta 0, pivote en su
    /// articulación. Para las partes que ya existen y para el nodo «grupo» (estirado). No toca nada si
    /// el nodo ya está donde dice la tabla (menos de 0,05 unidades), para no ensuciar el YAML.
    /// </summary>
    private static bool PlacePart(RectTransform rt, float[] rect, float[] pivotDown)
    {
        var w = rect[2] - rect[0];
        var h = rect[3] - rect[1];
        var minUp = new Vector2(rect[0], U - rect[3]);
        var box = BoxOf(rt);
        if (Near(box.Min, minUp) && Near(box.Size, new Vector2(w, h)) && Near(PivotDown(rt), Pt(pivotDown)))
        {
            return false;
        }

        var pb = BoxOf((RectTransform)rt.parent);
        var maxUp = new Vector2(rect[2], U - rect[1]);
        rt.anchorMin = new Vector2((minUp.x - pb.Min.x) / pb.Size.x, (minUp.y - pb.Min.y) / pb.Size.y);
        rt.anchorMax = new Vector2((maxUp.x - pb.Min.x) / pb.Size.x, (maxUp.y - pb.Min.y) / pb.Size.y);
        rt.pivot = new Vector2((pivotDown[0] - rect[0]) / w, (rect[3] - pivotDown[1]) / h);
        rt.offsetMin = Vector2.zero;
        rt.offsetMax = Vector2.zero;
        return true;
    }

    private static RectTransform NewRect(string name, Transform parent)
    {
        var go = new GameObject(name, typeof(RectTransform));
        var rect = (RectTransform)go.transform;
        rect.SetParent(parent, false);
        return rect;
    }

    /// <summary>La Image de un nodo nuevo: sin clics (RNF-02), apagada y sin sprite (no pinta un recuadro blanco).</summary>
    private static Image AddOffImage(RectTransform rt)
    {
        var image = rt.gameObject.AddComponent<Image>();
        image.raycastTarget = false;
        image.enabled = false;
        return image;
    }

    private static void AddNodes(Personaje c, List<string> log)
    {
        var path = PrefabPath(c);
        var root = PrefabUtility.LoadPrefabContents(path);
        try
        {
            var added = new List<string>();
            foreach (var n in c.nodos)
            {
                EnsureNode(root.transform, n, added);
            }

            var wired = EnsureFace(root, c, added);
            if (added.Count > 0 || wired)
            {
                PrefabUtility.SaveAsPrefabAsset(root, path);
                log.Add(c.prefab + ": " + added.Count + " añadidos (" + string.Join(", ", added.ToArray()) + ")");
            }
            else
            {
                log.Add(c.prefab + ": ya estaba completo, sin cambios");
            }
        }
        finally
        {
            PrefabUtility.UnloadPrefabContents(root);
        }
    }

    private static void EnsureNode(Transform root, Nodo n, List<string> added)
    {
        var parent = root.Find(n.padre) as RectTransform;
        if (parent == null)
        {
            throw new InvalidOperationException(root.name + ": no existe el padre «" + n.padre + "» de " + n.nombre);
        }

        var node = parent.Find(n.nombre) as RectTransform;
        var isNew = node == null;
        if (isNew)
        {
            node = NewRect(n.nombre, parent); // se añade como último hijo: Cuello/Ojos/Boca quedan después de Torso
            added.Add(n.nombre);
        }

        switch (n.tipo)
        {
            case "grupo":
                // Estirado como Cuerpo y Tronco de la familia; el pivote en «punto».
                if (isNew) PlacePart(node, n.rect, n.punto);
                break;

            case "imagen":
                if (isNew)
                {
                    AddOffImage(node);
                    PlacePoint(node, Pt(n.punto), n.rect);
                }

                break;

            case "articulacion":
                if (isNew) PlacePoint(node, Pt(n.punto), null); // pivote vacío en la articulación
                var segment = node.Find(n.imagen) as RectTransform;
                if (segment == null)
                {
                    segment = NewRect(n.imagen, node);
                    AddOffImage(segment);
                    PlacePoint(segment, Pt(n.punto), n.rect); // cuelga del pivote: girarlo gira el segmento
                    added.Add(n.imagen);
                }

                break;

            default:
                throw new InvalidOperationException("tipo de nodo desconocido «" + n.tipo + "» en " + n.nombre);
        }
    }

    /// <summary>CharacterFace en la raíz, con eyes/mouth apuntando a las Image de Ojos y Boca. Devuelve si cambió algo.</summary>
    private static bool EnsureFace(GameObject root, Personaje c, List<string> added)
    {
        var changed = false;
        var face = root.GetComponent<CharacterFace>();
        if (face == null)
        {
            face = root.AddComponent<CharacterFace>();
            added.Add("CharacterFace");
            changed = true;
        }

        var so = new SerializedObject(face);
        changed |= WireImage(so, "eyes", FindImage(root, c, "Ojos"));
        changed |= WireImage(so, "mouth", FindImage(root, c, "Boca"));
        so.ApplyModifiedPropertiesWithoutUndo();
        return changed;
    }

    private static Image FindImage(GameObject root, Personaje c, string nombre)
    {
        foreach (var n in c.nodos)
        {
            if (n.nombre == nombre)
            {
                var t = root.transform.Find(n.padre + "/" + nombre);
                return t == null ? null : t.GetComponent<Image>();
            }
        }

        return null;
    }

    /// <summary>Solo rellena la propiedad si está vacía: no pisa un cableado manual.</summary>
    private static bool WireImage(SerializedObject so, string property, Image image)
    {
        var prop = so.FindProperty(property);
        if (prop == null || image == null || prop.objectReferenceValue != null)
        {
            return false;
        }

        prop.objectReferenceValue = image;
        return true;
    }

    /// <summary>El PNG por nombre: primero en la carpeta del personaje, luego en sus subcarpetas. Lo pasa a Sprite Single.</summary>
    private static Sprite FindSprite(string carpeta, string nombre)
    {
        if (string.IsNullOrEmpty(nombre))
        {
            return null;
        }

        var folder = Art + "/" + carpeta;
        var path = folder + "/" + nombre + ".png";
        if (!File.Exists(path))
        {
            path = null;
            foreach (var guid in AssetDatabase.FindAssets(nombre + " t:Texture2D", new[] { folder }))
            {
                var candidate = AssetDatabase.GUIDToAssetPath(guid);
                if (Path.GetFileNameWithoutExtension(candidate) == nombre)
                {
                    path = candidate;
                    break;
                }
            }
        }

        if (path == null)
        {
            return null;
        }

        // Como ImportSprites de BuildRigs: con «Multiple» (el valor de fábrica) LoadAssetAtPath<Sprite> devuelve nulo.
        var importer = (TextureImporter)AssetImporter.GetAtPath(path);
        if (importer.textureType != TextureImporterType.Sprite || importer.spriteImportMode != SpriteImportMode.Single)
        {
            importer.textureType = TextureImporterType.Sprite;
            importer.spriteImportMode = SpriteImportMode.Single;
            importer.alphaIsTransparency = true;
            importer.mipmapEnabled = false;
            importer.SaveAndReimport();
        }

        return AssetDatabase.LoadAssetAtPath<Sprite>(path);
    }

    private static void ApplySprites(Personaje c, List<string> log)
    {
        var path = PrefabPath(c);
        var root = PrefabUtility.LoadPrefabContents(path);
        try
        {
            var done = new List<string>();
            var missing = new List<string>();

            // 1. Las partes que ya existen: sprite, rect y pivote. Primero ellas, porque sus hijos se anclan a su rect.
            if (c.partes != null)
            {
                foreach (var part in c.partes)
                {
                    var rt = root.transform.Find(part.ruta) as RectTransform;
                    var sprite = FindSprite(c.carpeta, part.sprite);
                    if (rt == null || sprite == null)
                    {
                        missing.Add(part.sprite);
                        continue; // lo que ya se ve se queda como está
                    }

                    var image = rt.GetComponent<Image>();
                    var changed = PlacePart(rt, part.rect, part.pivote);
                    if (image.sprite != sprite || !image.enabled)
                    {
                        image.sprite = sprite;
                        image.enabled = true;
                        changed = true;
                    }

                    if (changed) done.Add(part.nombre);
                }
            }

            // 2. Los nodos nuevos, padre antes que hijo. Un pivote se recoloca siempre (su padre pudo cambiar de rect).
            foreach (var n in c.nodos)
            {
                var parent = root.transform.Find(n.padre) as RectTransform;
                var node = parent == null ? null : parent.Find(n.nombre) as RectTransform;
                if (node == null)
                {
                    missing.Add(n.nombre + " (¿corrió «nodos»?)");
                    continue;
                }

                RectTransform target = node;
                if (n.tipo == "articulacion")
                {
                    PlacePoint(node, Pt(n.punto), null);
                    target = node.Find(n.imagen) as RectTransform;
                }
                else if (n.tipo == "grupo")
                {
                    PlacePart(node, n.rect, n.punto);
                    continue;
                }

                var image = target == null ? null : target.GetComponent<Image>();
                var sprite = FindSprite(c.carpeta, n.sprite);
                if (image == null || sprite == null)
                {
                    if (n.tipo == "articulacion" && target != null) PlacePoint(target, Pt(n.punto), n.rect);
                    missing.Add(n.sprite);
                    continue; // sin PNG: la Image queda apagada
                }

                PlacePoint(target, Pt(n.punto), n.rect);
                image.sprite = sprite;
                image.enabled = true;
                done.Add(target.name);
            }

            var faced = ApplyFaceSet(root, c, done);

            // 3. Algoritm: con sus partes puestas, el sprite entero de Cuerpo sobra (se vería doble).
            if (c.guia && AllEnabled(root, new[] { T + "/Torso", T + "/BrazoIzq", T + "/BrazoDer", C + "/PiernaIzq", C + "/PiernaDer" }))
            {
                var body = root.transform.Find(C).GetComponent<Image>();
                if (body != null && body.enabled)
                {
                    body.enabled = false;
                    done.Add("Cuerpo apagado");
                }
            }

            if (done.Count > 0 || faced)
            {
                PrefabUtility.SaveAsPrefabAsset(root, path);
            }

            log.Add(c.prefab + ": " + done.Count + " aplicados" + (done.Count > 0 ? " (" + string.Join(", ", done.ToArray()) + ")" : "")
                + (missing.Count > 0 ? " | SIN PNG: " + string.Join(", ", missing.ToArray()) : ""));
        }
        finally
        {
            PrefabUtility.UnloadPrefabContents(root);
        }
    }

    private static bool AllEnabled(GameObject root, string[] paths)
    {
        foreach (var p in paths)
        {
            var t = root.transform.Find(p);
            var image = t == null ? null : t.GetComponent<Image>();
            if (image == null || !image.enabled || image.sprite == null)
            {
                return false;
            }
        }

        return true;
    }

    /// <summary>Crea o actualiza &lt;prefijo&gt;_cara.asset con los sprites de cara que existan y lo asigna a faceSet.</summary>
    private static bool ApplyFaceSet(GameObject root, Personaje c, List<string> done)
    {
        var found = new Dictionary<string, Sprite>();
        foreach (var pair in FaceNames)
        {
            var sprite = FindSprite(c.carpeta, c.prefijo + "_" + pair[1]);
            if (sprite != null) found[pair[0]] = sprite;
        }

        if (found.Count == 0)
        {
            return false;
        }

        var assetPath = Art + "/" + c.carpeta + "/" + c.prefijo + "_cara.asset";
        var set = AssetDatabase.LoadAssetAtPath<CharacterFaceSet>(assetPath);
        if (set == null)
        {
            set = ScriptableObject.CreateInstance<CharacterFaceSet>();
            AssetDatabase.CreateAsset(set, assetPath);
        }

        var so = new SerializedObject(set);
        foreach (var pair in found)
        {
            var prop = so.FindProperty("<" + pair.Key + ">k__BackingField");
            if (prop != null) prop.objectReferenceValue = pair.Value; // solo lo que existe: no vacía lo ya asignado
        }

        so.ApplyModifiedPropertiesWithoutUndo();
        EditorUtility.SetDirty(set);

        var changed = false;
        var face = root.GetComponent<CharacterFace>();
        if (face != null)
        {
            var faceSo = new SerializedObject(face);
            var prop = faceSo.FindProperty("faceSet");
            if (prop != null && prop.objectReferenceValue != set)
            {
                prop.objectReferenceValue = set;
                faceSo.ApplyModifiedPropertiesWithoutUndo();
                changed = true;
            }
        }

        done.Add(c.prefijo + "_cara.asset (" + found.Count + " sprites)");
        return changed;
    }

    {
        var root = AssetDatabase.LoadAssetAtPath<GameObject>(PrefabPath(c));
        if (root == null)

            + " | CharacterFace " + (face == null ? "no" : "sí") + " | segmentado " + (IsSegmented(root.transform) ? "sí" : "no"));
    }

    // ================================================================ modo «clips»
    //
    // Reescribe las curvas de los .anim existentes (ClearCurves + SetEditorCurve sobre el asset
    // cargado: GUID, controlador y eventos intactos). Cada clip lleva rotación en TODOS los huesos del
    // personaje: un hueso que un clip no usa se queda en su pose de reposo y no vuelve a la pose en T
    // del prefab (writeDefaultValues = true).
    //
    // Signos (el eje Z de Unity gira en sentido antihorario con valores positivos; la pantalla es
    // el espejo del personaje que mira al jugador):
    //   Brazo izquierdo/derecho (hombro): + en BrazoDer y − en BrazoIzq ABREN y SUBEN el brazo
    //     (elevación). Al revés, el brazo se cierra contra el cuerpo. Con el brazo derecho en reposo
    //     (cuelga hacia abajo y afuera), +42 lo deja casi horizontal (Point).
    //   Codo: la flexión «hacia arriba» (la mano sube hacia el hombro) es − en CodoIzq y + en CodoDer,
    //     o sea, el mismo signo que la elevación de su lado. El pliegue «hacia dentro» (manos a la
    //     cintura) es el contrario. En el código se dan en grados de flexión y Sym los reparte.
    //   Pierna: − en PiernaIzq y + en PiernaDer ABREN el muslo (rodillas hacia fuera).
    //   Rodilla: + en RodillaIzq y − en RodillaDer devuelven la pantorrilla HACIA DENTRO.
    //   Cuerpo/Tronco/Cuello/Cabeza: + inclina a la izquierda de la pantalla, − a la derecha. La
    //     inclinación de Walk/Run («body lean») es negativa: hacia donde mira el personaje.
    //
    // «Sym(izq, der, t, v...)»: v > 0 abre/sube/flexiona hacia arriba; Izq recibe −v y Der +v. Para
    // las rodillas, v < 0 es hacia dentro. «Rot» da el valor tal cual (se le suma el reposo).
    //
    // Posición: solo en Lienzo y Lienzo/Cuerpo, más las de Tronco y Piernas que BuildRigs ya ponía
    // para agacharse y levantar el pie (son del prefab de cada miembro de la familia, que tiene su
    // propio controlador). Las tres formas de Algoritm comparten controlador y prefabs distintos:
    // allí la única posición está en Cuerpo.
    //
    // «segmentado» NO está en el JSON: se deduce del prefab (la Image de AntepiernaIzq con sprite y
    // encendida). Con las piernas partidas, agacharse NO escala las piernas: baja el tronco, abre el
    // muslo, devuelve la pantorrilla y la acorta con escala Y de Rodilla (CharacterRig_DA131_
    // ConExtremidadesPartidas… lo vigila). Con el arte actual (no segmentado) se conserva el truco de
    // BuildRigs: la pierna entera escala en Y y las curvas de rodilla existen pero no se ven.

    /// <summary>Lo que cada clip necesita saber del personaje.</summary>
    private sealed class Ctx
    {
        public string Id; public bool Guide; public bool Segmented;
        public float K = 1f; public float A = 1f;
        public float Leg;    // de la cadera al suelo
        public float Thigh;  // de la cadera a la rodilla
        public float Shin;   // de la rodilla al suelo
        public float ElbowRest;
    }

    private static void WriteClips(Tabla tabla, List<string> log)
    {
        var guideDone = false;
        foreach (var p in tabla.personajes)
        {
            Ctx x;
            string error;
            if (!ReadContext(p, out x, out error))
            {
                log.Add(p.prefab + ": " + error);
                continue;
            }

            if (p.guia)
            {
                if (guideDone) continue; // las tres formas comparten los nueve clips
                guideDone = true;
                WriteSpecs(Art + "/Algoritm/Animations", GuideClips(x), log, "Algoritm (3 formas)");
            }
            else
            {
                WriteSpecs(Art + "/" + Family[p.id].Folder + "/Animations", HumanClips(x), log, p.prefab + (x.Segmented ? " [segmentado]" : ""));
            }
        }
    }

    private static bool ReadContext(Personaje p, out Ctx x, out string error)
    {
        x = null;
        error = null;
        var go = AssetDatabase.LoadAssetAtPath<GameObject>(PrefabPath(p));
        if (go == null)
        {
            error = "no existe el prefab";
            return false;
        }

        var root = go.transform;
        var needed = p.guia ? new[] { LE, RE, LK, RK, T + "/Torso", T + "/Ojos" } : new[] { LE, RE, LK, RK, HD, NK + "/Cabeza/Ojos" };
        foreach (var path in needed)
        {
            if (root.Find(path) == null)
            {
                error = "falta «" + path + "»: corra primero Execute(\"nodos\")";
                return false;
            }
        }

        x = new Ctx { Id = p.id, Guide = p.guia };
        if (p.guia)
        {
            return true;
        }

        var info = Family[p.id];
        x.K = info.Tempo;
        x.A = info.Amp;
        x.ElbowRest = info.ElbowRest;
        x.Segmented = IsSegmented(root);

        var hip = PivotDown((RectTransform)root.Find(LL));
        var floor = PivotDown((RectTransform)root.Find(C)).y; // el pivote de Cuerpo está en el suelo
        var knee = PivotDown((RectTransform)root.Find(LK)).y;
        x.Leg = Mathf.Max(floor - hip.y, 1f);
        x.Thigh = Mathf.Clamp(knee - hip.y, 0.25f * x.Leg, 0.75f * x.Leg);
        x.Shin = x.Leg - x.Thigh;
        return true;
    }

    /// <summary>Segmentado = la antepierna ya se dibuja (Image con sprite y encendida).</summary>
    private static bool IsSegmented(Transform root)
    {
        var shin = root.Find(LK + "/AntepiernaIzq");
        var image = shin == null ? null : shin.GetComponent<Image>();
        return image != null && image.sprite != null && image.enabled;
    }

    private sealed class Curve
    {
        public string Path; public string Prop;
        public readonly List<Vector2> Keys = new List<Vector2>(); // x = tiempo, y = valor

        public Type BindingType { get { return Prop == Alpha ? typeof(CanvasGroup) : typeof(RectTransform); } }

        public void Set(float t, float v)
        {
            for (var i = 0; i < Keys.Count; i++)
            {
                if (Mathf.Abs(Keys[i].x - t) < 1e-4f)
                {
                    Keys[i] = new Vector2(Keys[i].x, v);
                    return;
                }

                if (Keys[i].x > t)
                {
                    Keys.Insert(i, new Vector2(t, v));
                    return;
                }
            }

            Keys.Add(new Vector2(t, v));
        }
    }

    private sealed class Spec
    {
        public readonly string Action;
        public readonly string File;
        public readonly float Length;
        public readonly float K;
        public readonly bool Loop;
        private readonly Dictionary<string, float> _rest;
        private readonly string[] _bones;
        private readonly Dictionary<string, Curve> _curves = new Dictionary<string, Curve>();
        private readonly List<Curve> _order = new List<Curve>();

        public Spec(string action, string file, float baseLength, float k, bool loop, Dictionary<string, float> rest, string[] bones)
        {
            Action = action; File = file; K = k; Length = baseLength * k; Loop = loop; _rest = rest; _bones = bones;
        }

        public IList<Curve> Curves { get { return _order; } }

        private Curve Get(string path, string prop, bool create = true)
        {
            Curve curve;
            var key = path + "|" + prop;
            if (!_curves.TryGetValue(key, out curve))
            {
                if (!create) return null;
                curve = new Curve { Path = path, Prop = prop };
                _curves[key] = curve;
                _order.Add(curve);
            }

            return curve;
        }

        private float RestOf(string path)
        {
            float v;
            return _rest.TryGetValue(path, out v) ? v : 0f;
        }

        private float BaseOf(string path, string prop)
        {
            if (prop == RotZ) return RestOf(path);
            return prop == ScaleX || prop == ScaleY ? 1f : 0f;
        }

        /// <summary>Rotación en Z: pares (tiempo base, grados) SUMADOS a la pose de reposo del hueso. Los tiempos se multiplican por el tempo.</summary>
        public Spec Rot(string path, params float[] tv)
        {
            var c = Get(path, RotZ);
            for (var i = 0; i + 1 < tv.Length; i += 2) c.Set(tv[i] * K, tv[i + 1] + RestOf(path));
            return this;
        }

        /// <summary>Cualquier otra propiedad, con el valor absoluto.</summary>
        public Spec Raw(string path, string prop, params float[] tv)
        {
            var c = Get(path, prop);
            for (var i = 0; i + 1 < tv.Length; i += 2) c.Set(tv[i] * K, tv[i + 1]);
            return this;
        }

        /// <summary>Un movimiento simétrico: v &gt; 0 abre, sube o flexiona hacia arriba (Izq recibe −v, Der +v).</summary>
        public Spec Sym(string left, string right, params float[] tv)
        {
            var l = Get(left, RotZ);
            var r = Get(right, RotZ);
            for (var i = 0; i + 1 < tv.Length; i += 2)
            {
                l.Set(tv[i] * K, RestOf(left) - tv[i + 1]);
                r.Set(tv[i] * K, RestOf(right) + tv[i + 1]);
            }

            return this;
        }

        /// <summary>
        /// Squash &amp; stretch de Cuerpo: pares (tiempo, escala Y). X compensa (1 − 0,5·(Y − 1)): con Y
        /// entre 0,92 y 1,08 la X queda entre 1,04 y 0,96, y ninguna pasa del 15 % (DA §13.2).
        /// </summary>
        public Spec Vol(params float[] tv)
        {
            var y = Get(C, ScaleY);
            var xs = Get(C, ScaleX);
            for (var i = 0; i + 1 < tv.Length; i += 2)
            {
                y.Set(tv[i] * K, tv[i + 1]);
                xs.Set(tv[i] * K, 1f - 0.5f * (tv[i + 1] - 1f));
            }

            return this;
        }

        public Spec Drop(string path, string prop)
        {
            var key = path + "|" + prop;
            Curve c;
            if (_curves.TryGetValue(key, out c))
            {
                _curves.Remove(key);
                _order.Remove(c);
            }

            return this;
        }

        /// <summary>
        /// Movimiento secundario: la curva «dst» (rotación) copia la de «src» con unos cuadros de
        /// retraso (2 a 4 en la cabeza) y una ganancia. Reemplaza lo que «dst» tuviera. En clips en
        /// bucle el retraso da la vuelta; en los otros se sujeta a los extremos. «bias» resta una
        /// media a la fuente (p. ej. la mitad de la altura de la flotación).
        /// </summary>
        public Spec Follow(string src, string srcProp, string dst, float frames, float gain, float bias = 0f)
        {
            var source = Get(src, srcProp, false);
            if (source == null || source.Keys.Count == 0)
            {
                throw new InvalidOperationException(Action + ": Follow necesita primero la curva " + src + " " + srcProp);
            }

            Close(source);
            var curve = Build(source.Keys);
            var target = Get(dst, RotZ);
            target.Keys.Clear();
            var delay = frames / Fps;
            var from = BaseOf(src, srcProp);
            var to = RestOf(dst);
            foreach (var t in SampleTimes(source.Keys))
            {
                var tt = Loop ? Mathf.Repeat(t - delay, Length) : Mathf.Clamp(t - delay, 0f, Length);
                target.Set(t, to + gain * (curve.Evaluate(tt) - from - bias));
            }

            return this;
        }

        private List<float> SampleTimes(List<Vector2> keys)
        {
            var times = new List<float> { 0f };
            for (var i = 0; i < keys.Count; i++)
            {
                if (i > 0) times.Add(0.5f * (keys[i - 1].x + keys[i].x));
                times.Add(keys[i].x);
            }

            times.Add(Length);
            return times.Where(t => t >= -1e-4f && t <= Length + 1e-4f).Select(t => Mathf.Clamp(t, 0f, Length)).Distinct().OrderBy(t => t).ToList();
        }

        /// <summary>En los bucles, que la curva empiece en 0 y acabe en Length (si falta, con el valor del otro extremo).</summary>
        private void Close(Curve c)
        {
            if (!Loop || c.Keys.Count == 0) return;
            if (c.Keys[0].x > 1e-4f) c.Keys.Insert(0, new Vector2(0f, c.Keys[c.Keys.Count - 1].y));
            if (c.Keys[c.Keys.Count - 1].x < Length - 1e-4f) c.Keys.Add(new Vector2(Length, c.Keys[0].y));
        }

        /// <summary>
        /// Último paso: todo hueso sin rotación se queda en su pose de reposo (anti pose en T), cada
        /// curva cubre [0, Length] y, en los bucles, acaba donde empieza. Devuelve los avisos.
        /// </summary>
        public List<string> Finish()
        {
            var warnings = new List<string>();
            foreach (var bone in _bones)
            {
                if (Get(bone, RotZ, false) == null)
                {
                    Get(bone, RotZ).Set(0f, RestOf(bone));
                }
            }

            foreach (var c in _order)
            {
                if (c.Keys.Count == 1) c.Set(Length, c.Keys[0].y);
                if (Loop)
                {
                    Close(c);
                    var gap = c.Keys[c.Keys.Count - 1].y - c.Keys[0].y;
                    if (c.Prop == RotZ) gap = Mathf.Repeat(gap + 180f, 360f) - 180f; // una vuelta entera (Spin) sí cierra
                    if (Mathf.Abs(gap) > 1e-3f)
                    {
                        warnings.Add(Action + " " + c.Path + " " + c.Prop + ": el bucle no cierra (" + c.Keys[0].y + " → " + c.Keys[c.Keys.Count - 1].y + ")");
                    }
                }
                else
                {
                    if (c.Keys[0].x > 1e-4f) c.Keys.Insert(0, new Vector2(0f, c.Keys[0].y));
                    if (c.Keys[c.Keys.Count - 1].x < Length - 1e-4f) c.Keys.Add(new Vector2(Length, c.Keys[c.Keys.Count - 1].y));
                }

                foreach (var k in c.Keys)
                {
                    if (k.x < -1e-4f || k.x > Length + 1e-4f) warnings.Add(Action + " " + c.Path + " " + c.Prop + ": clave fuera de [0, Length] en t=" + k.x);
                }
            }

            return warnings;
        }
    }

    /// <summary>AnimationCurve con tangentes ClampedAuto, como en BuildRigs.</summary>
    private static AnimationCurve Build(List<Vector2> keys)
    {
        var frames = new Keyframe[keys.Count];
        for (var i = 0; i < keys.Count; i++) frames[i] = new Keyframe(keys[i].x, keys[i].y);
        var curve = new AnimationCurve(frames);
        for (var i = 0; i < curve.length; i++)
        {
            AnimationUtility.SetKeyLeftTangentMode(curve, i, AnimationUtility.TangentMode.ClampedAuto);
            AnimationUtility.SetKeyRightTangentMode(curve, i, AnimationUtility.TangentMode.ClampedAuto);
        }

        return curve;
    }

    private static void WriteSpecs(string folder, List<Spec> specs, List<string> log, string who)
    {
        var written = 0;
        foreach (var spec in specs)
        {
            foreach (var warning in spec.Finish()) log.Add("AVISO " + who + ": " + warning);
            if (WriteClip(folder, spec, log)) written++;
        }

        log.Add(who + ": " + written + "/" + specs.Count + " clips reescritos en " + folder);
    }

    private static bool WriteClip(string folder, Spec spec, List<string> log)
    {
        var path = folder + "/" + spec.File + ".anim";
        var clip = AssetDatabase.LoadAssetAtPath<AnimationClip>(path);
        if (clip == null)
        {
            log.Add("FALTA el clip " + path + ": no se crea (no se tocan ni controladores ni GUID)");
            return false;
        }

        var frameRate = clip.frameRate;
        var events = AnimationUtility.GetAnimationEvents(clip);
        var settings = AnimationUtility.GetAnimationClipSettings(clip);
        clip.ClearCurves();
        clip.frameRate = frameRate > 0f ? frameRate : Fps;
        foreach (var c in spec.Curves)
        {
            var binding = EditorCurveBinding.FloatCurve(c.Path, c.BindingType, c.Prop);
            AnimationUtility.SetEditorCurve(clip, binding, Build(c.Keys));
        }

        if (settings.loopTime != spec.Loop)
        {
            log.Add("AVISO " + spec.File + ": loopTime del asset (" + settings.loopTime + ") difiere del diseñado (" + spec.Loop + "); se conserva el del asset");
        }

        settings.stopTime = spec.Length; // loopTime, eventos y fotogramas por segundo se conservan
        AnimationUtility.SetAnimationClipSettings(clip, settings);
        AnimationUtility.SetAnimationEvents(clip, events);
        EditorUtility.SetDirty(clip);
        return true;
    }

    {

    /// <summary>La cabeza sigue al cuerpo con 2 y 4 cuadros de retraso: el movimiento secundario.</summary>
    private static void HeadLags(Spec s, string srcPath, string srcProp, float neckGain, float headGain, float bias = 0f)
    {
        s.Follow(srcPath, srcProp, NK, 2, neckGain, bias);
        s.Follow(srcPath, srcProp, HD, 4, headGain, bias);
    }

    /// <summary>
    /// Agacharse a una profundidad. Devuelve la bajada efectiva del tronco (unidades del lienzo) para
    /// que el clip baje Tronco lo mismo. «tw» son pares (tiempo base, peso 0..1).
    /// No segmentado: las piernas bajan y se acortan con escala Y (el truco de BuildRigs), con el
    /// muslo abierto «thighFlat» grados. Segmentado: el muslo se abre «thighSeg» grados, la
    /// pantorrilla vuelve hacia dentro «kneeRatio» veces eso y se acorta con escala Y de la rodilla
    /// lo justo para que el pie siga en el suelo; si hiciera falta menos de 0,4 se baja menos.
    /// </summary>
    private static float Crouch(Ctx x, Spec s, float depthFrac, float thighFlat, float thighSeg, float kneeRatio, params float[] tw)
    {
        const float minShin = 0.4f;
        var depth = depthFrac * x.Leg;
        if (x.Segmented)
        {
            var a = thighSeg * Mathf.Deg2Rad;
            var net = (thighSeg - thighSeg * kneeRatio) * Mathf.Deg2Rad;
            depth = Mathf.Max(0f, Mathf.Min(depth, x.Leg - x.Thigh * Mathf.Cos(a) - minShin * x.Shin * Mathf.Cos(net)));
        }

        for (var i = 0; i + 1 < tw.Length; i += 2)
        {
            var t = tw[i];
            var w = tw[i + 1];
            var d = depth * w;
            s.Raw(LL, PosY, t, -d);
            s.Raw(RL, PosY, t, -d);
            if (!x.Segmented)
            {
                var ratio = (x.Leg - d) / x.Leg;
                s.Raw(LL, ScaleY, t, ratio);
                s.Raw(RL, ScaleY, t, ratio);
                if (thighFlat > 0f) s.Sym(LL, RL, t, thighFlat * w);
            }
            else
            {
                var open = thighSeg * w;
                var back = open * kneeRatio;
                var cosOpen = Mathf.Cos(open * Mathf.Deg2Rad);
                var cosNet = Mathf.Cos((open - back) * Mathf.Deg2Rad);
                var shin = Mathf.Clamp((x.Leg - d - x.Thigh * cosOpen) / (x.Shin * cosNet), minShin, 1f);
                s.Sym(LL, RL, t, open);
                s.Sym(LK, RK, t, -back);
                s.Raw(LK, ScaleY, t, shin);
                s.Raw(RK, ScaleY, t, shin);
            }
        }

        return depth;
    }

    /// <summary>Respiración de 3,2 s: pecho y hombros suben al inspirar, el peso pasa de una pierna a otra, la cabeza llega tarde.</summary>
    private static Spec Idle(Ctx x)
    {
        var a = x.A;
        var s = NewSpec(x, "Idle", "idle", 3.2f);
        s.Vol(0, 1f, 1.28f, 1f + 0.010f * a, 3.2f, 1f);
        s.Rot(T, 0, 0, 0.8f, 0.8f * a, 1.6f, 0, 2.4f, -0.8f * a, 3.2f, 0);
        s.Sym(LA, RA, 0, 0, 1.28f, 1.5f * a, 3.2f, 0);
        s.Sym(LE, RE, 0, 0, 1.28f, 1.0f * a, 3.2f, 0);
        s.Sym(LL, RL, 0, 0, 0.8f, 0.6f * a, 1.6f, 0, 2.4f, -0.6f * a, 3.2f, 0);
        s.Rot(NK, 0, 0, 1.0f, 0.9f, 2.2f, -0.6f, 3.2f, 0);

        return s;
    }

    /// <summary>
    /// Caminar y correr (misma estructura que BuildRigs, con inclinación, rebote, codos, rodillas y cabeza).
    /// Los pasos alternan: la pierna izquierda sube en 0,25·p y la derecha en 0,75·p.
    /// </summary>
    private static Spec Walk(Ctx x, string action, string file, float p, float amp, float armSwing, float lean, float sway, float flexLo, float flexHi, float squash)
    {
        var lift = 0.14f * x.Leg * amp;
        var mid = 0.5f * (flexLo + flexHi);
        var s = NewSpec(x, action, file, p);
        s.Raw(LL, PosY, 0, 0, 0.25f * p, lift, 0.5f * p, 0, p, 0);
        s.Raw(RL, PosY, 0, 0, 0.5f * p, 0, 0.75f * p, lift, p, 0);
        s.Rot(LL, 0, 0, 0.25f * p, -4f * amp, 0.5f * p, 0, p, 0);
        s.Rot(RL, 0, 0, 0.5f * p, 0, 0.75f * p, 4f * amp, p, 0);
        // La rodilla del pie que vuela se dobla un poco hacia dentro (+ en Izq, − en Der).
        s.Rot(LK, 0, 0, 0.25f * p, 6f * amp, 0.5f * p, 0, p, 0);
        s.Rot(RK, 0, 0, 0.5f * p, 0, 0.75f * p, -6f * amp, p, 0);
        if (x.Segmented)
        {
            s.Raw(LK, ScaleY, 0, 1, 0.25f * p, 0.9f, 0.5f * p, 1, p, 1);
            s.Raw(RK, ScaleY, 0, 1, 0.5f * p, 1, 0.75f * p, 0.9f, p, 1);
        }

        s.Raw(C, PosY, 0, 0, 0.25f * p, 10f * amp, 0.5f * p, 0, 0.75f * p, 10f * amp, p, 0);
        s.Rot(C, 0, lean, 0.25f * p, lean - sway, 0.5f * p, lean, 0.75f * p, lean + sway, p, lean);
        s.Vol(0, 1f - squash, 0.25f * p, 1f + 0.7f * squash, 0.5f * p, 1f - squash, 0.75f * p, 1f + 0.7f * squash, p, 1f - squash);
        s.Rot(T, 0, 0, 0.25f * p, 1.2f * amp, 0.5f * p, 0, 0.75f * p, -1.2f * amp, p, 0);
        s.Rot(LA, 0, 0, 0.25f * p, armSwing, 0.5f * p, 0, 0.75f * p, -armSwing, p, 0);
        s.Rot(RA, 0, 0, 0.25f * p, armSwing, 0.5f * p, 0, 0.75f * p, -armSwing, p, 0);
        s.Rot(LE, 0, -mid, 0.25f * p, -flexHi, 0.5f * p, -mid, 0.75f * p, -flexLo, p, -mid);
        s.Rot(RE, 0, mid, 0.25f * p, flexLo, 0.5f * p, mid, 0.75f * p, flexHi, p, mid);
        HeadLags(s, C, RotZ, -0.35f, -0.2f);
        return s;
    }

    /// <summary>Recoger: un respingo de anticipación, se agacha, alcanza, se levanta con lo recogido (1,4 s).</summary>
    private static Spec PickUp(Ctx x)
    {

    private static Dictionary<string, float> GuideRest()
    {
        var rest = new Dictionary<string, float>();
        foreach (var bone in BonesGuide) rest[bone] = 0f;
        return rest;
    }

    private static Spec GuideSpec(string action, string file, float length, bool loop = true)
    {
        return new Spec(action, "char_algoritm_anim_" + file, length, 1f, loop, GuideRest(), BonesGuide);
    }

    /// <summary>
    /// Piernas que cuelgan: siguen la altura del cuerpo con 3 cuadros de retraso (juntas arriba,
    /// abiertas abajo) y las rodillas devuelven lo que abre el muslo; los brazos suben un poco con
    /// 4 cuadros de retraso, los codos siguen al brazo y el tronco, al giro del cuerpo. «mean» es la
    /// altura media de la flotación.
    /// </summary>
    private static void Dangle(Spec s, float mean)
    {
        DangleLegs(s, mean);
        s.Follow(C, PosY, LA, 4, -0.4f, mean);
        s.Follow(C, PosY, RA, 4, 0.4f, mean);
        s.Follow(LA, RotZ, LE, 3, 0.8f);
        s.Follow(RA, RotZ, RE, 3, 0.8f);
        s.Follow(C, RotZ, T, 2, 0.4f);
    }

    private static void DangleLegs(Spec s, float mean)
    {
        s.Follow(C, PosY, LL, 3, 0.5f, mean);
        s.Follow(C, PosY, RL, 3, -0.5f, mean);
        s.Follow(LL, RotZ, LK, 3, -0.8f);
        s.Follow(RL, RotZ, RK, 3, -0.8f);
    }
